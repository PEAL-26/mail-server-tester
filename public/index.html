POST /api/smtp/test

  -H "Content-Type: application/json" \
  -d '{
    "host": "smtp.exemplo.com",
    "port": 587,
    "security": "TLS",
    "user": "usuario@exemplo.com",
    "pass": "suasenha",
    "from": "de@exemplo.com",
    "to": "para@exemplo.com",
    "subject": "Teste SMTP"
  }'
